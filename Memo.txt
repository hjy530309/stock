-- ======================================================================
-- 1) 사용자 / 인증
-- ======================================================================
CREATE TABLE users (
    num             NUMBER PRIMARY KEY,
    name            VARCHAR2(50) NOT NULL,
    email           VARCHAR2(100) NOT NULL UNIQUE,
    password        VARCHAR2(200) NOT NULL,
    role            VARCHAR2(20) DEFAULT 'user' NOT NULL,
    phone           VARCHAR2(30),
    notify_stock    NUMBER(1) DEFAULT 1 NOT NULL CHECK (notify_stock IN (0, 1)),
    notify_comment  NUMBER(1) DEFAULT 1 NOT NULL CHECK (notify_comment IN (0, 1))
);

CREATE TABLE saved_email_token (
    token       VARCHAR2(100) PRIMARY KEY,
    email       VARCHAR2(100) NOT NULL,
    expires_at  TIMESTAMP NOT NULL
);

-- ======================================================================
-- 2) 종목 / 관심종목 / 알림
-- ======================================================================
CREATE TABLE stock (
    id          NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    stock_code  VARCHAR2(20) NOT NULL UNIQUE,
    stock_name  VARCHAR2(100),
    full_name   VARCHAR2(200),
    market      VARCHAR2(30),
    updated_at  TIMESTAMP DEFAULT SYSTIMESTAMP
);

CREATE TABLE watch_list (
    id          NUMBER PRIMARY KEY,
    user_num    NUMBER NOT NULL,
    stock_code  VARCHAR2(20) NOT NULL,
    stock_name  VARCHAR2(100),
    created_at  TIMESTAMP DEFAULT SYSTIMESTAMP,
    CONSTRAINT uq_watch_list_user_stock UNIQUE (user_num, stock_code),
    CONSTRAINT fk_watch_list_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

CREATE TABLE stock_alert (
    id           NUMBER PRIMARY KEY,
    user_num     NUMBER NOT NULL,
    stock_code   VARCHAR2(20) NOT NULL,
    stock_name   VARCHAR2(100),
    change_rate  VARCHAR2(20),
    alert_type   VARCHAR2(10),
    price        VARCHAR2(20),
    alert_read   NUMBER(1) DEFAULT 0 NOT NULL CHECK (alert_read IN (0, 1)),
    created_at   TIMESTAMP DEFAULT SYSTIMESTAMP,
    CONSTRAINT fk_stock_alert_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

-- ======================================================================
-- 3) 뉴스
-- ======================================================================
CREATE TABLE news_data (
    link            VARCHAR2(1000) PRIMARY KEY,
    category        VARCHAR2(200),
    title           VARCHAR2(500),
    summary         CLOB,
    sentiment       VARCHAR2(30),
    pub_date        VARCHAR2(30),
    clickbait_prob  NUMBER(6,2),
    article_type    VARCHAR2(50),
    type_prob       NUMBER(6,2),
    created_at      TIMESTAMP DEFAULT SYSTIMESTAMP
);

CREATE TABLE news_data_sec (
    link            VARCHAR2(1000) PRIMARY KEY,
    category        VARCHAR2(200),
    title           VARCHAR2(500),
    summary         CLOB,
    sentiment       VARCHAR2(30),
    pub_date        VARCHAR2(30),
    clickbait_prob  NUMBER(6,2),
    article_type    VARCHAR2(50),
    type_prob       NUMBER(6,2),
    created_at      TIMESTAMP DEFAULT SYSTIMESTAMP
);

CREATE TABLE news_stock_map (
    id          NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    link        VARCHAR2(1000) NOT NULL,
    stock_code  VARCHAR2(20) NOT NULL,
    stock_name  VARCHAR2(100),
    match_score NUMBER(10,6)
);

CREATE TABLE news_impact (
    id          NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    link        VARCHAR2(1000) NOT NULL,
    stock_code  VARCHAR2(20) NOT NULL,
    base_price  NUMBER(18,4),
    price_30m   NUMBER(18,4),
    impact_30m  NUMBER(10,4),
    created_at  TIMESTAMP DEFAULT SYSTIMESTAMP
);

CREATE TABLE news_likes (
    like_id     NUMBER PRIMARY KEY,
    news_link   VARCHAR2(1000) NOT NULL,
    user_num    NUMBER NOT NULL,
    created_at  TIMESTAMP DEFAULT SYSTIMESTAMP,
    CONSTRAINT uq_news_likes UNIQUE (news_link, user_num),
    CONSTRAINT fk_news_likes_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

CREATE TABLE news_comments (
    comment_id   NUMBER PRIMARY KEY,
    news_link    VARCHAR2(1000) NOT NULL,
    user_num     NUMBER NOT NULL,
    content      VARCHAR2(2000) NOT NULL,
    created_at   TIMESTAMP DEFAULT SYSTIMESTAMP,
    CONSTRAINT fk_news_comments_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

-- ======================================================================
-- 4) 커뮤니티
-- ======================================================================
CREATE TABLE community_board (
    board_id     NUMBER PRIMARY KEY,
    user_num     NUMBER NOT NULL,
    category     VARCHAR2(30) NOT NULL,
    title        VARCHAR2(200) NOT NULL,
    news_link    VARCHAR2(1000),
    content      CLOB NOT NULL,
    view_count   NUMBER DEFAULT 0 NOT NULL,
    like_count   NUMBER DEFAULT 0 NOT NULL,
    created_at   DATE DEFAULT SYSDATE NOT NULL,
    updated_at   DATE DEFAULT SYSDATE NOT NULL,
    CONSTRAINT fk_board_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

CREATE TABLE community_comment (
    comment_id   NUMBER PRIMARY KEY,
    board_id     NUMBER NOT NULL,
    user_num     NUMBER NOT NULL,
    content      VARCHAR2(1000) NOT NULL,
    created_at   DATE DEFAULT SYSDATE NOT NULL,
    updated_at   DATE DEFAULT SYSDATE NOT NULL,
    CONSTRAINT fk_comment_board FOREIGN KEY (board_id) REFERENCES community_board(board_id) ON DELETE CASCADE,
    CONSTRAINT fk_comment_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

CREATE TABLE community_like (
    like_id      NUMBER PRIMARY KEY,
    board_id     NUMBER NOT NULL,
    user_num     NUMBER NOT NULL,
    created_at   DATE DEFAULT SYSDATE NOT NULL,
    CONSTRAINT uq_community_like UNIQUE (board_id, user_num),
    CONSTRAINT fk_community_like_board FOREIGN KEY (board_id) REFERENCES community_board(board_id) ON DELETE CASCADE,
    CONSTRAINT fk_community_like_user FOREIGN KEY (user_num) REFERENCES users(num) ON DELETE CASCADE
);

CREATE TABLE community_tag (
    tag_id       NUMBER PRIMARY KEY,
    tag_name     VARCHAR2(100) NOT NULL UNIQUE
);

CREATE TABLE community_board_tag (
    board_id     NUMBER NOT NULL,
    tag_id       NUMBER NOT NULL,
    CONSTRAINT pk_community_board_tag PRIMARY KEY (board_id, tag_id),
    CONSTRAINT fk_community_board_tag_board FOREIGN KEY (board_id) REFERENCES community_board(board_id) ON DELETE CASCADE,
    CONSTRAINT fk_community_board_tag_tag FOREIGN KEY (tag_id) REFERENCES community_tag(tag_id) ON DELETE CASCADE
);

-- ======================================================================
-- 5) 자산
-- ======================================================================
CREATE TABLE spending_data (
    spend_id          NUMBER PRIMARY KEY,
    user_id           NUMBER NOT NULL,
    month             NUMBER(2) NOT NULL,
    year              NUMBER(4) NOT NULL,
    transaction_date  DATE NOT NULL,
    amount            NUMBER(15,2) NOT NULL,
    vendor            VARCHAR2(200) NOT NULL,
    category          VARCHAR2(100),
    CONSTRAINT fk_spending_user FOREIGN KEY (user_id) REFERENCES users(num) ON DELETE CASCADE
);

CREATE TABLE asset_analysis_history (
    analysis_id               NUMBER PRIMARY KEY,
    user_id                   NUMBER NOT NULL,
    current_asset             NUMBER(18,2),
    monthly_income            NUMBER(18,2),
    monthly_expense           NUMBER(18,2),
    monthly_saving            NUMBER(18,2),
    goal_amount               NUMBER(18,2),
    goal_months               NUMBER(5),
    expected_return           NUMBER(10,4),
    age                       NUMBER(3),
    job_type                  VARCHAR2(50),
    risk_preference           VARCHAR2(50),
    prediction                NUMBER(2),
    prediction_label          VARCHAR2(50),
    tone_title                VARCHAR2(100),
    model_prediction          NUMBER(2),
    model_prediction_label    VARCHAR2(50),
    model_probability         NUMBER(10,6),
    required_monthly_saving   NUMBER(18,2),
    estimated_final_asset     NUMBER(18,2),
    goal_gap                  NUMBER(18,2),
    message                   CLOB,
    created_at                DATE DEFAULT SYSDATE NOT NULL,
    CONSTRAINT fk_asset_history_user FOREIGN KEY (user_id) REFERENCES users(num) ON DELETE CASCADE
);

-- ======================================================================
-- 6) 시퀀스
-- ======================================================================
CREATE SEQUENCE user_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE community_board_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE community_comment_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE community_tag_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE community_like_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE news_likes_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE news_comments_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE news_impact_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE watch_list_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE stock_alert_seq START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE seq_spending_id START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE asset_analysis_history_seq START WITH 1 INCREMENT BY 1 NOCACHE;

-- ======================================================================
-- 7) 인덱스
-- ======================================================================
CREATE INDEX idx_community_board_user_created ON community_board(user_num, created_at);
CREATE INDEX idx_community_board_category ON community_board(category);
CREATE INDEX idx_community_comment_board_created ON community_comment(board_id, created_at);
CREATE INDEX idx_community_like_board ON community_like(board_id);
CREATE INDEX idx_community_like_user ON community_like(user_num);
CREATE INDEX idx_community_board_tag_tag ON community_board_tag(tag_id);

CREATE INDEX idx_news_data_pub_date ON news_data(pub_date);
CREATE INDEX idx_news_data_category ON news_data(category);
CREATE INDEX idx_news_data_sec_pub_date ON news_data_sec(pub_date);
CREATE INDEX idx_news_data_sec_category ON news_data_sec(category);
CREATE INDEX idx_news_impact_link_created ON news_impact(link, created_at);
CREATE INDEX idx_news_likes_link ON news_likes(news_link);
CREATE INDEX idx_news_comments_link_created ON news_comments(news_link, created_at);

CREATE INDEX idx_watch_list_user ON watch_list(user_num);
CREATE INDEX idx_stock_alert_user_read_created ON stock_alert(user_num, alert_read, created_at);

CREATE INDEX idx_spending_user_year_month_date ON spending_data(user_id, year, month, transaction_date);
CREATE INDEX idx_asset_history_user_created ON asset_analysis_history(user_id, created_at);
